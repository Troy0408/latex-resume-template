\newcommand{\contactfield}[2]{%
  \ExpandArgs{o}\IfBlankF{#1}{%
    \nextcontactsep\mbox{#2}\let\nextcontactsep\contactsep
  }}
\newcommand{\resumeidentity}{%
  \raggedright
  {\fontsize{23pt}{26pt}\selectfont\bfseries\color{fgcolor}\resumename\par}%
  \def\nextcontactsep{\vspace{6pt}\noindent}%
  \contactfield\resumephone{\phone{\resumephone}}%
  \contactfield\resumeemail{\email{\resumeemail}}%
  \contactfield\resumeaddress{\basicInfoItem{\faLocationDot}{\resumeaddress}}%
  \contactfield\githubuser{\github{\githubuser}}%
  \contactfield\resumewebsite{\website[\resumewebsitetext]{\resumewebsite}}\par}
